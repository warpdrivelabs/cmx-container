① 全景：端到端回存链路

前端 ChangeSetCollector.export() # 增量协议：仅回传变更，见 §2 ▼ saveDocData(host, def, { saveMode, changes | snapshot, collector? }) ▼ POST /api/doc/save?domain&application&module&file ▼ handler doc_save： │ resolve_doc_meta（读定义 + base，带缓存）→ DocMetaView │ run_validation（事务外二次校验，error 阻断） # §14.2 │ save_ctx(ctx) → SaveCtx { actor_id, actor_name, doc_file, op_override } ▼ DocSaver::save(mm, db_id, meta, mode, changes, sctx) │ begin_with_guard → txn_id │ apply_and_version(txn_id)： # 事务内一体，任一步失败即回滚 │ ├ apply_merge | apply_replace # 批量写 §5 + 审计 §6 + 乐观锁 §8 │ ├ record_versions # 版本快照 §7（versioning 开启时） │ └ new_root_baselines # 算回传基线 §8 │ commit ▼ SaveResult { ok, mode, affected, updatedAt } ▼ 前端：collector.reset() + refreshBaselines(updatedAt)
四期能力叠加，全部通用、零业务规则：第一期批量 SQL（消 N+1）· C审计填充 · B1版本快照 · B2乐观锁 · F批量端点。每一层都对任意 L1..Ln 单据自动生效。

② 增量协议（changeset）

前端 ChangeSetCollector 订阅协调器各层 CmxDataSet 事件，累积「按层」的最小变更集——只回传变了的东西。

{
  "cv_header": {                                // 层 key：表名 / 层 id / 嵌套全路径
    "inserted": [ { id, upper_id?, line_no?, fields:{...整行} } ],
    "updated":  [ { id, fields:{仅变更列}, baseline: update_time } ],
    "deleted":  [ "id1", "id2" ]              // 裸 id 数组
  }
}
桶内容净化规则
inserted整行 fields + 结构列新增又删 → 净零抵消，不回传
updated仅变更列 + baseline（乐观锁基线，§8）updated 只带用户改过的键（脏列集）
deleted裸 id 数组删的若是新增行 → 从 inserted 移除，不进 deleted
设计优点：① 净零抵消 —— 新增后又删除的行不回后端。② updated 只传变更列（稀疏），省体积。③ 按层分组，后端按 layer_order 拓扑序精确写。层 key 兼容三形态：表名 / 层 id / 嵌套全路径（前端 collector 用点分路径，如 cv_batch.cv_header.cv_acc_line）。

③ 两种模式：merge / replace

模式语义写法适用
merge（默认）按 changeset 精确 INSERT(UPSERT)/UPDATE/DELETE，保主键/审计，写入最小父先（layer_order 正序）UPSERT+UPDATE → 子先（逆序）DELETE常规编辑保存
replace对 rootId 子树先 DELETE 旧行、再全量 INSERT snapshot（前端免 diff）子先（逆序）子查询链删子树 → 父先全量插整单覆盖 / 版本回滚

replace 的子查询链删除（方案 E，第一期）

沿 upper_id 链圈定 rootId 子树：每层一条 DELETE，WHERE 用嵌套子查询上溯到根层，零预 SELECT、无 id 物化（避免并发漂移），往返数 = 层数。childKey 由元数据相邻推导（与装载器同源，独立于 relation 命名）。

④ 事务内一体流水线

核心是 DocSaver::save（saver.rs）：一个事务 guard 包住「写入 + 记版本 + 算基线」，任一步失败整体回滚——快照失败绝不 commit。

pub async fn save(mm, db_id, meta, mode, changes, sctx) -> Result<SaveResult> {
    let guard = ctx.begin_with_guard(db_id).await?;   // 开事务
    let txn_id = guard.txn_id();
    let outcome = Self::apply_and_version(mm, db_id, txn_id, meta, mode, changes, sctx).await;
    match outcome {
        Ok((affected, updated_at)) => { guard.commit().await?; Ok(SaveResult{..}) }
        Err(e) => { guard.rollback().await; Err(e) }        // 全回滚
    }
}

// apply_and_version：事务内一体（供单单 save 与批量 save_batch 共用，不含 begin/commit）
async fn apply_and_version(...) -> Result<(u64, Vec<UpdatedBaseline>)> {
    let audit = AuditCtx { actor: sctx.actor_id, now: Utc::now() };  // 全程共用一个时间戳
    let affected = match mode { Merge => apply_merge(...), Replace => apply_replace(...) }?;
    Self::record_versions(...).await?;              // B1
    let updated_at = new_root_baselines(...);        // B2
    Ok((affected, updated_at))
}

⑤ 批量 SQL（消除 N+1，第一期）

旧实现逐行 INSERT ... ON CONFLICT，1000 行 = 1000+ 往返。第一期改为同层多行合并：

操作SQL分组
UPSERT / INSERTINSERT INTO t (cols) VALUES ($1..),($n..) ON CONFLICT(id) DO UPDATE按「列集合」分组（各行 fields 稀疏）
UPDATEUPDATE t SET c=v.c FROM (VALUES ...) AS v(id,cols) WHERE t.id=v.id按「变更列集合」分组
DELETEDELETE FROM t WHERE id = ANY($1)批量（本就是）

参数上限 60000 分批；SQL 构造抽为纯函数 build_multi_insert_sql / build_multi_update_sql（可单测）。1000 行分录：1000+ 往返 → O(层数×列集组) 条 SQL。

正确性兜底（通用，不涉业务）

  • H1 静默零写防护：assert_all_keys_matched —— changeset 每个 key 必须对上某层，否则报错，杜绝「path 漂移导致保存成功却一行没写」。
  • H2 对账：INSERT+UPDATE 严格对账（期望行数 == 实际 affected），实际 < 期望即报错回滚（id 不存在/并发删）；DELETE 幂等不纳入。
  • 防注入一致性：写 SQL 全部 quote_ident 双引号包裹表/列名，列撞关键字也安全；列名过 schema 白名单。

⑥ 审计字段自动填充（方案 C，第二期）

保存时由服务端权威填充审计列，忽略/覆盖前端传值——审计不可伪造。

路径填充列
INSERT（新增/UPSERT）create_by=actor, create_time=now, update_by=actor, update_time=now, delete_flag=0
UPDATE强制注入 update_by=actor, update_time=now（覆盖前端传值）
ON CONFLICT DO UPDATE SET排除 create_by/create_time（创建审计不可变，撞已存在 id 时不冲掉原创建人）
顺带修一个潜在 bug：create_by/create_time/delete_flag 是 NOT NULL 无默认，而前端 _insertedRow 不回传技术列 → 纯新增行 INSERT 本会因 NOT NULL 失败。后端填充补上此缺口。
actor 兜底 0，永不阻断：create_by/update_by 是 BIGINT，而 auth_context.user_id 是 String（系统身份为字面量 "system"）。缺失/空/非数字 → 兜底 0（约定 0=系统）。保存永不因身份缺失失败，兼容认证白名单/未配置认证服务。now 全事务共用一个 Utc::now()。

⑦ 版本快照（方案 B1，第三期）

单据开启 versioning 时，每次 save 在同一事务内、写入后 commit 前：重装载整单 → 列式快照 → 追加一版（append-only 台账 cmx_doc_revision）。

record_versions(txn_id)： if !meta.versioning_enabled() { return } # 仅开启时记（定义 moduleMeta.versioning.enabled） 收集受影响 rootId + op（inserted→create / updated→update） for each root： DocQuery { layers:{ 根层:{ filter:{ id: rootId } } } } DocLoader::load_txn(txn_id = Some(..)) # 事务内重装载，看得到未提交写！ ColumnarCodec::encode → 列式快照 JSONB DocRevision::record（旧版 is_current=0，插新版 rev_no+1，actor/op 落台账）
within-txn 重装载是关键：DocLoader::load 默认走连接池（txn_id=None）看不到未提交写；快照必须用 load_txn(Some(txn_id)) 在同事务连接上重装载，才能反映「保存后」的整单终态。副作用：事务内遇缺失兄弟表会污染整个 PG 事务，故非主兄弟查询套 SAVEPOINT，失败回滚到 savepoint 再跳过。

配套读端点（已存在）：GET /api/doc/revisions（版本时间线）、GET /api/doc/revision（取某版快照）、POST /api/doc/restore（把某历史版恢复为新当前版，走 replace 模式，op_override="restore"）。

⑧ 乐观锁（方案 B2，第三期）

并发下防「后保存者静默覆盖先保存者」（lost update）。仅根层加锁（单据级并发防护）。

机制

1. 前端 export() 每 updated 行带 baseline = row.update_time # 装载时的旧值（readonly 列，未被用户改） 2. 后端根层 UPDATE 加谓词： UPDATE t SET ..., update_time = <新now> # SET 写新值（方案 C） FROM (VALUES (id, cols.., baseline)) AS v(id, cols.., __oplock) WHERE t.id = v.id AND t.update_time = v.__oplock # WHERE 比旧值 3. 基线陈旧的行不匹配 → 不更新 → affected < expected 4. 冲突判定：对该组 id 补 SELECT —— id 存在=冲突(409) / 不存在=行不存在(H2) 5. save 成功回传 updatedAt（每根新 update_time），前端 refreshBaselines 刷新
WHERE 比旧值、SET 写新值——两者分离，天然正确：baseline 是装载时的 update_time（WHERE 用它检测并发），apply_audit_update 把新 now 写进 SET。二者互不干扰。
冲突 = HTTP 409：新增 BizError::Conflict → cmx_api_types::Error::Conflict → 409。classify_update_shortfall 补一次 SELECT id WHERE id=ANY($缺失)：命中=冲突（"单据已被他人修改，请刷新后重试"），未命中=沿用 H2「行不存在」业务错。前端 saveDocData 见 409 抛 err.conflict=true。
基线精度安全：装载路径把 timestamptz 读为全精度 DateTime<Utc> → to_rfc3339()（保微秒）→ 回传绑定同类型。实测 update_time = '…123456+00:00'::timestamptz 等值匹配为真，DB→前端→DB 往返逐字节可比。注：只有服务端权威 readonly 列（update_time）能当基线；用户可编辑列不行（无原值快照）。

⑨ 批量单据端点（方案 F，第四期）

一次请求保存多单。一批可混多种单据定义（每单自带坐标）。POST /api/doc/save/batch。

事务边界语义失败行为
atomic: true（默认）N 单共用一个大事务任一单失败 → 整批回滚，返回 Err（带出错单 index，冲突仍 409），无部分提交
atomic: false每单独立事务（复用单单 save）逐单成败互不影响，全收进 results（失败单 ok=false + error），永不整体 Err

C/B1/B2 对每单自动生效——批量只是外层事务编排，不改单单语义。错误经 from_batch_item 加「第 N 单」前缀且保留变体（Conflict 仍映射 409）。

// 请求体
{
  "atomic": true,
  "docs": [
    { domain, application, module, file, saveMode?, changes | snapshot },
    ...
  ]
}
// 响应
{ atomic, count, results: [ { index, ok, mode, affected, updatedAt?, error? } ] }

⑪ 前端调用（collector / save）

ChangeSetCollector — 累积变更

import { ChangeSetCollector, saveDocData } from 'cmx-data-comp'

const cs = new ChangeSetCollector(ms).attach()   // ms = CmxMasterSlave；装载后 attach
// ... 用户编辑（增/改/删行）...
cs.isDirty()                    // 是否有未保存变更
const changes = cs.export()      // { path: { inserted, updated(带baseline), deleted } }
方法作用
attach()订阅协调器各层 CmxDataSet 事件（装载后调）
export()导出最小 changeset；updated 行带 baseline: row.update_time（B2）
reset()清空累积（保存成功后调）
refreshBaselines(updatedAt)用后端回传新 update_time 刷新行基线（B2，支持连续保存不刷新页）

saveDocData — 单单保存

try {
  const result = await saveDocData(host, def, {
    saveMode: 'merge',          // 'merge' | 'replace'
    changes,                     // merge 用 changes；replace 用 snapshot
    collector: cs,               // 传则成功后自动 refreshBaselines(result.updatedAt)
    signal: ac.signal
  })
  cs.reset()                     // { ok, mode, affected, updatedAt }
} catch (e) {
  if (e.conflict) alert('单据已被他人修改，请刷新后重试')  // HTTP 409（B2）
  else throw e
}

def 同 loadDocData 的坐标（domain/application/module/file/dbId），加可选 saveApiPath 覆盖端点。dbId 走 db_id 请求头。响应兼容「带信封 {code,msg,data}」与「门户拆信封的裸 {ok,mode,affected}」两形态。

saveDocDataBatch — 批量保存

const r = await saveDocDataBatch(host, {
  atomic: true,                // true=一事务全成全败；false=逐单隔离
  docs: [
    { domain, application, module, file, saveMode:'merge', changes: cs1.export() },
    { domain, application, module, file, saveMode:'replace', snapshot: {...} }
  ],
  collector: cs,               // 逐单回传 updatedAt 都刷新基线
  dbId: 'fico'
})
// r = { atomic, count, results:[{index,ok,mode,affected,updatedAt?,error?}] }
// atomic 模式任一单冲突 → 抛 err.conflict=true（整批回滚）

⑫ 常用调用配方

A. 标准编辑保存（merge + 乐观锁 + 基线刷新）

const cs = new ChangeSetCollector(ms).attach()
// 用户编辑后：
if (cs.isDirty()) {
  await saveDocData(host, def, { saveMode:'merge', changes: cs.export(), collector: cs })
  cs.reset()
}

B. 整单覆盖（replace）

const snapshot = rootDs.exportColumnar({ includeChildren: true })  // 整树列式包
await saveDocData(host, def, { saveMode:'replace', snapshot })

C. 批量过账/导入（atomic 全成全败）

const docs = selectedDocs.map(d => ({
  domain:'fi', application:'cmxfico', module:'gl', file:d.file,
  saveMode:'merge', changes: d.collector.export()
}))
try {
  const r = await saveDocDataBatch(host, { atomic:true, docs, dbId:'fico' })
  // 全部成功：r.results 全 ok=true
} catch (e) {
  if (e.conflict) alert('批量中有单据被他人修改，整批已回滚')
}

D. 批量导入（非 atomic，容忍部分失败）

const r = await saveDocDataBatch(host, { atomic:false, docs, dbId:'fico' })
const failed = r.results.filter(x => !x.ok)
if (failed.length) reportFailures(failed)  // 每项带 index + error；成功单已提交
版本化开启：给单据定义 moduleMeta 加 "versioning": { "enabled": true } 即启用版本快照（B1）。关闭时 save 不写 cmx_doc_revision——通用存储层不强制版本化，由定义控制。
